#
# Language support.
#
#   index        the name of every message string, one per line, in order;
#                the single source of truth for the string numbering
#   <lang>.l     the text of every string in one language
#   langcomp.c   compiles <lang>.l into the binary file Services loads
#
# langstrs.h (the LANG_* / message index constants the code uses) is
# generated from `index` here, so the two can never disagree.
#

set(lang_out "${PROJECT_BINARY_DIR}/lang")
file(MAKE_DIRECTORY "${lang_out}")

# --- langstrs.h ------------------------------------------------------------
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/index")
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/index" lang_names)

set(defines "")
set(names "")
set(i 0)
foreach(string_name IN LISTS lang_names)
  string(LENGTH "${string_name}" len)
  math(EXPR pad "32 - ${len}")
  if(pad LESS 1)
    set(pad 1)
  endif()
  string(REPEAT " " ${pad} spaces)
  string(APPEND defines "#define ${string_name} ${spaces}${i}\n")
  string(APPEND names "    \"${string_name}\",\n")
  math(EXPR i "${i} + 1")
endforeach()

# file(CONFIGURE) only rewrites the header when its content changes, so a
# reconfigure does not rebuild the whole tree.
file(CONFIGURE OUTPUT "${lang_out}/langstrs.h" @ONLY CONTENT
"/* langstrs.h -- generated by CMake from lang/index.  Do not edit. */
${defines}
#define NUM_BASE_STRINGS ${i}

#ifdef LANGSTR_ARRAY
static const char * const base_langstrs[] = {
${names}};
#endif
")

# --- langcomp and the compiled languages -----------------------------------
add_executable(langcomp langcomp.c)
target_link_libraries(langcomp PRIVATE services_config)

file(GLOB lang_sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.l")
list(SORT lang_sources)

# langcomp reads `index` and <lang>.l from its working directory and writes
# <lang> beside them, so both are staged into the build tree first.
add_custom_command(OUTPUT "${lang_out}/index"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
          "${CMAKE_CURRENT_SOURCE_DIR}/index" "${lang_out}/index"
  DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/index"
  VERBATIM)

set(lang_outputs "")
foreach(source IN LISTS lang_sources)
  get_filename_component(lang "${source}" NAME_WE)
  add_custom_command(OUTPUT "${lang_out}/${lang}"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${source}" "${lang}.l"
    COMMAND $<TARGET_FILE:langcomp> "${lang}.l"
    WORKING_DIRECTORY "${lang_out}"
    DEPENDS langcomp "${source}" "${lang_out}/index"
    COMMENT "Compiling language ${lang}"
    VERBATIM)
  list(APPEND lang_outputs "${lang_out}/${lang}")
endforeach()

add_custom_target(languages ALL DEPENDS ${lang_outputs})
install(FILES ${lang_outputs} DESTINATION "${SERVICES_DATA_DIR}/languages")
